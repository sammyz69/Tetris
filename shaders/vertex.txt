#version 330 core

layout (location=0) in vec3 vertexPos;
layout (location=1) in int colorIndex;

out vec3 fragmentColor;
out vec2 fragmentTexCoord;

const vec3 colors[8] = vec3[](
    vec3(1.0, 0.0, 0.0),
    vec3(0.0, 1.0, 0.0),
    vec3(0.0, 0.0, 1.0),
    vec3(0.25, 0.25, 0.25),
    vec3(0.5, 0.5, 0.5),
    vec3(0.75, 0.75, 0.75),
    vec3(0.1333, 0.6941, 0.2980),
    vec3(1.0, 1.0, 1.0)
);

void main()
{
    gl_Position = vec4(vertexPos, 1.0);
    fragmentColor = colors[colorIndex];
    // Map NDC [-1,1] -> UV [0,1] with Y flip so texture appears upright
    fragmentTexCoord = vec2(
        (vertexPos.x + 1.0) * 0.5,
        (1.0 - vertexPos.y) * 0.5
    );
}
